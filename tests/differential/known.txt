# Known, intentional differences between zav and libclamav 1.5.4, each with its reason. zav-diff
# check reports these without failing, and fails when a listed difference stops occurring, so
# this list is exact. Format: CASE sN|sigN zav=RESULT  # reason
#
# BYTE[x-y]HEXSIG: libclamav finds the leading byte only within 34 - min(length, 8) bytes of the
# signature, whatever y says; zav applies the range as written.
hex-bracket-before-far s2 zav=Diff.Bracket.Far2 Diff.Bracket.Far4 Diff.Bracket.Far8
hex-bracket-before-far s3 zav=Diff.Bracket.Far2 Diff.Bracket.Far4 Diff.Bracket.Far8
hex-bracket-before-far s4 zav=Diff.Bracket.Far2 Diff.Bracket.Far4 Diff.Bracket.Far8
hex-bracket-before-far s5 zav=Diff.Bracket.Far2 Diff.Bracket.Far4 Diff.Bracket.Far8
# ~aa and ~a? are documented for functionality level 240; libclamav 1.5 does not parse them yet.
hex-negated-nibble sig1 zav=LOADS
hex-negated-nibble sig2 zav=LOADS
# YARA: zav follows yara (tools/diff/yara_check.py checks zav against it), and libclamav's YARA
# support differs from yara here:
# - it refuses a hex string without two literal bytes in a row, which yara accepts;
yara-hex sig7 zav=LOADS
# - it ignores the regex modifiers i and s;
yara-regex s2 zav=YARA.RegexNocase
yara-regex s3 zav=YARA.RegexDotall
# - it misreads hex literals of 2^31 and up and has no uint16be;
yara-reads s1 zav=YARA.ReadUint16Be YARA.ReadUint32 YARA.ReadUint8
# - it loads xor strings and never matches them; zav refuses xor over all 256 keys and expands
#   a range, as yara does;
yara-xor sig1 zav=REJECTS
yara-xor s2 zav=YARA.XorRange
yara-xor s3 zav=YARA.XorRange
# - it loads rules without strings and never fires them (yara fires them); zav refuses them.
yara-no-strings sig1 zav=REJECTS
# Outside zav's YARA subset (modules, for loops, entrypoint, one-byte strings): refused with a
# reason. pe.* is refused by yara too unless the module is imported.
yara-unsupported sig1 zav=REJECTS
yara-unsupported sig2 zav=REJECTS
yara-unsupported sig3 zav=REJECTS
yara-unsupported sig4 zav=REJECTS
